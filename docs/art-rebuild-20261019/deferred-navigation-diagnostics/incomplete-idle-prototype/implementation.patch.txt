+ * Remaining traversable steps to any destination, independent of this turn's
+ * budget, occupants and control zones. This is navigation intent, not permission
+ * to move: the move field still owns legal paths and their terrain costs.
+ */
+export function computeNavigationDistances(
+  rules: MovementRules,
+  state: GameState,
+  movement: MovementClass,
+  destinations: readonly Coord[],
+): ReadonlyMap<number, number> {
+  const board = boardOf(rules, state);
+  const battlefield = new Battlefield(state, rules.content);
+  const distances = new Map<number, number>();
+  const queue: Array<{ at: Coord; distance: number }> = [];
+  for (const destination of destinations) {
+    if (!board.contains(destination) || !battlefield.cell(destination).admits(movement)) continue;
+    const index = board.indexOf(destination);
+    if (distances.has(index)) continue;
+    distances.set(index, 0);
+    queue.push({ at: destination, distance: 0 });
+  }
+  for (const { at: destination, distance } of queue) {
+    for (const from of board.neighbours(destination)) {
+      const index = board.indexOf(from);
+      if (distances.has(index) || !battlefield.cell(from).admits(movement)) continue;
+      // Climbing and dropping need not be symmetric: walk the reverse graph.
+      if (battlefield.traversalCost(from, destination, movement) === null) continue;
+      distances.set(index, distance + 1);
+      queue.push({ at: from, distance: distance + 1 });
+    }
+  }
+  return distances;
